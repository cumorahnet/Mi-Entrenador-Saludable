// ── Visibility change ──────────────────────────────────────────────────── useEffect(() => { const handle = () => { if (document.hidden) { // Si la página está oculta (cambio de ventana/pestaña) if (statusRef.current === 'running') { // Y el entrenamiento estaba en marcha pausedAtRef.current = Date.now(); // Guardar el momento de la pausa setStatus('paused'); // Poner el estado en pausado } releaseWake(); // Liberar el wake lock para que la pantalla pueda apagarse } else { // Si la página vuelve a estar visible if (statusRef.current === 'paused' && pausedAtRef.current !== null) { // Y estaba pausado por visibilidad // Ajustar orígenes de tiempo por el tiempo que estuvo oculto const hiddenMs = Date.now() - pausedAtRef.current; if (workoutStartRef.current !== null) workoutStartRef.current += hiddenMs; if (stepStartRef.current !== null) stepStartRef.current += hiddenMs; pausedAtRef.current = null; // Limpiar el momento de pausa setStatus('running'); // Reanudar el entrenamiento } requestWake(); // Solicitar el wake lock de nuevo } }; document.addEventListener('visibilitychange', handle); return () => document.removeEventListener('visibilitychange', handle); }, [requestWake, releaseWake]);
